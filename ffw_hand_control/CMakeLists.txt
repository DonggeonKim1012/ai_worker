cmake_minimum_required(VERSION 3.16)
project(ffw_hand_control)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(urdf REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(hand_control_bridge src/hand_control_bridge.cpp)
target_compile_features(hand_control_bridge PRIVATE cxx_std_17)
target_compile_options(hand_control_bridge PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(hand_control_bridge rclcpp std_msgs std_srvs sensor_msgs trajectory_msgs rcl_interfaces urdf)
target_link_libraries(hand_control_bridge nlohmann_json::nlohmann_json)
install(TARGETS hand_control_bridge DESTINATION lib/${PROJECT_NAME})
ament_package()
